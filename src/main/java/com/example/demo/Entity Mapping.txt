CREATE VIEW AvailableRoomTypes AS
SELECT
    rt.room_type_id AS roomTypeId,
    rt.description,
    rs.price,
    rs.quantity,
    s.season_start_date,
    s.season_end_date,
    h.hotel_Id AS hotelId,
    h.name AS hotelName,
    rt.capacity,
    rs.room_season_id AS roomSeasonId
FROM room_type rt
JOIN room_season rs ON rt.room_type_id = rs.room_type_id
JOIN season s ON rs.season_id = s.season_id
JOIN contract c ON s.contract_id = c.contract_id
JOIN hotel h ON c.hotel_id = h.hotel_id
WHERE s.season_end_date >= CURRENT_DATE;

WITH SeasonsCovered AS (
    SELECT s.start_date, s.end_date
    FROM Season s
    WHERE (s.start_date <= :checkInDate AND s.end_date >= :checkInDate)
        OR (s.start_date <= :checkOutDate AND s.end_date >= :checkOutDate)
),
GapExists AS (
    SELECT 1
    FROM SeasonsCovered sc1
    WHERE NOT EXISTS (
        SELECT 1
        FROM SeasonsCovered sc2
        WHERE sc2.start_date = sc1.end_date + INTERVAL 1 DAY
    )
    AND sc1.end_date < :checkInDate
),
AvailableHotels AS (
    SELECT h.hotel_id
    FROM Hotel h
    JOIN RoomSeason rs ON h.hotel_id = rs.hotel_id
    JOIN Room r ON rs.room_id = r.room_id
    WHERE r.available_rooms > 0
    AND rs.start_date <= :checkOutDate
    AND rs.end_date >= :checkInDate
)
SELECT h.*
FROM Hotel h
WHERE h.hotel_id IN (SELECT hotel_id FROM AvailableHotels)
AND NOT EXISTS (SELECT 1 FROM GapExists);
