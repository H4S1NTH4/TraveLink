 public Set<RoomType> getRoomTypes() {
        Set<RoomType> roomTypes = new HashSet<>();
        for (RoomSeason roomSeason : roomSeasons) {
            roomTypes.add(roomSeason.getRoomType());
        }
        return roomTypes;
